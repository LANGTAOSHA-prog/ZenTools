© 2026 ZenTools · 所有处理均在浏览器本地完成 const dropZone = document.getElementById('dropZone'); const fileInput = document.getElementById('fileInput'); const convertBtn = document.getElementById('convertBtn'); const downloadBtn = document.getElementById('downloadBtn'); const formatSelect = document.getElementById('formatSelect'); const fileList = document.getElementById('fileList'); let files = []; let convertedBlobs = []; dropZone.addEventListener('click', () => fileInput.click()); dropZone.addEventListener('dragover', (e) => { e.preventDefault(); dropZone.classList.add('dragover'); }); dropZone.addEventListener('dragleave', () => dropZone.classList.remove('dragover')); dropZone.addEventListener('drop', (e) => { e.preventDefault(); dropZone.classList.remove('dragover'); handleFiles(e.dataTransfer.files); }); fileInput.addEventListener('change', (e) => handleFiles(e.target.files)); function handleFiles(selectedFiles) { files = Array.from(selectedFiles).filter(f => f.type.startsWith('image/')); if (!files.length) return; renderFileList(); convertBtn.disabled = false; convertedBlobs = []; } function renderFileList() { fileList.innerHTML = files.map((f, i) => `
${f.name}等待转换
`).join(''); } convertBtn.addEventListener('click', async () => { convertBtn.disabled = true; convertedBlobs = []; for (let i = 0; i < files.length; i++) { const item = fileList.querySelector(`[data-index="${i}"] .status`); const format = formatSelect.value; try { const img = await loadImage(files[i]); const canvas = document.createElement('canvas'); canvas.width = img.width; canvas.height = img.height; const ctx = canvas.getContext('2d'); if (format === 'jpg') ctx.fillStyle = '#fff'; else ctx.fillStyle = 'transparent'; ctx.fillRect(0, 0, canvas.width, canvas.height); ctx.drawImage(img, 0, 0); const blob = await new Promise(r => canvas.toBlob(r, 'image/' + format, 0.9)); convertedBlobs.push({ blob, name: files[i].name.replace(/\.[^.]+$/, '') + '.' + format }); item.textContent = '✓ 转换成功'; item.classList.add('success'); } catch (e) { item.textContent = '✗ 转换失败'; item.classList.add('error'); } } downloadBtn.disabled = convertedBlobs.length === 0; convertBtn.disabled = false; }); function loadImage(file) { return new Promise((resolve, reject) => { const img = new Image(); img.onload = () => resolve(img); img.onerror = reject; img.src = URL.createObjectURL(file); }); } downloadBtn.addEventListener('click', () => { if (!convertedBlobs.length) return; if (convertedBlobs.length === 1) { const a = document.createElement('a'); a.href = URL.createObjectURL(convertedBlobs[0].blob); a.download = convertedBlobs[0].name; a.click(); } else { import('https://cdn.jsdelivr.net/npm/jszip@3.10.1/dist/jszip.min.js').then(JSZip => { const zip = new JSZip.default(); convertedBlobs.forEach(b => zip.file(b.name, b.blob)); zip.generateAsync({ type: 'blob' }).then(content => { const a = document.createElement('a'); a.href = URL.createObjectURL(content); a.download = 'converted-images.zip'; a.click(); }); }); } });
